```python
_CHARACTERISTIC_CALLS = {"roots", "extrema", "intersections"}
_ALLOWED_CALLS = _ALLOWED_CALLS | _CHARACTERISTIC_CALLS


def _validate_characteristic_call(node: ast.Call, line_no: int) -> None:
    name = node.func.id
    expected = 5 if name == "intersections" else 4
    if node.keywords:
        raise EngSyntaxError(f"line {line_no}: {name} keyword arguments are unsupported")
    if len(node.args) != expected:
        raise EngSyntaxError(
            f"line {line_no}: {name} expects {expected} positional arguments"
        )
    variable_index = 2 if name == "intersections" else 1
    if not isinstance(node.args[variable_index], ast.Name):
        raise EngSyntaxError(
            f"line {line_no}: {name} variable must be a symbolic identifier"
        )


def _validate_characteristic_statement_context(
    expression: ast.Expression,
    target: str | None,
    line_no: int,
) -> None:
    characteristic_calls = [
        child for child in ast.walk(expression)
        if isinstance(child, ast.Call)
        and isinstance(child.func, ast.Name)
        and child.func.id in _CHARACTERISTIC_CALLS
    ]
    if not characteristic_calls:
        return
    root = expression.body
    root_is_characteristic = (
        isinstance(root, ast.Call)
        and isinstance(root.func, ast.Name)
        and root.func.id in _CHARACTERISTIC_CALLS
    )
    if target is not None or not root_is_characteristic or len(characteristic_calls) != 1:
        raise EngSyntaxError(
            f"line {line_no}: characteristic analysis must be a standalone statement"
        )
```

Call `_validate_characteristic_call(...)` from `_validate_normal_node` before generic call recursion, and call `_validate_characteristic_statement_context(...)` after assignment parsing and AST validation but before constructing `ParsedStatement`.

```python
@dataclass(frozen=True)
class AnalysisDomain:
    lower_symbolic: sp.Expr
    upper_symbolic: sp.Expr
    lower_quantity: Any
    upper_quantity: Any
    unit: Any


def normalize_analysis_domain(context, lower_expression, upper_expression) -> AnalysisDomain:
    _, lower = context.evaluate_symbolic(sp.sympify(lower_expression))
    _, upper = context.evaluate_symbolic(sp.sympify(upper_expression))
    lower, upper = context.normalize_plot_bounds(lower, upper)
    lower_mag = float(lower.magnitude)
    upper_mag = float(upper.magnitude)
    if not math.isfinite(lower_mag) or not math.isfinite(upper_mag):
        raise EngEvaluationError("characteristic domain bounds must be finite")
    if not lower_mag < upper_mag:
        raise EngEvaluationError("characteristic domain requires lower < upper")
    return AnalysisDomain(
        lower_symbolic=sp.sympify(lower_expression),
        upper_symbolic=sp.sympify(upper_expression),
        lower_quantity=lower,
        upper_quantity=upper,
        unit=lower.units,
    )
```

Catch Pint `DimensionalityError` and convert it to an EngCalc domain diagnostic instead of leaking Pint internals.

- [ ] **Step 4: Implement exact finite root solving and candidate validation**

Use two exact attempts:

```python
def _exact_real_solution_set(expression, variable):
    solution_set = sp.solveset(sp.Eq(expression, 0), variable, domain=sp.S.Reals)
    if isinstance(solution_set, sp.FiniteSet):
        return tuple(solution_set), False
    if solution_set is sp.S.EmptySet:
        return (), False
    if solution_set is sp.S.Reals:
        return (), False
    try:
        solved = sp.solve(sp.Eq(expression, 0), variable)
    except (NotImplementedError, ValueError):
        return (), True
    if solved and all(not getattr(item, "free_symbols", set()) - expression.free_symbols for item in solved):
        return tuple(solved), False
    return (), True
```

Accept a root only when the evaluated magnitude is zero within the existing EngCalc zero policy; exact `sp.simplify(expression.subs(variable, candidate)) == 0` is preferred.
